<!-- Micro-interaction script for password toggle -->
<script>
        document.querySelectorAll('button[type="button"]').forEach(btn => {
            btn.addEventListener('click', function() {
                const input = this.parentElement.querySelector('input');
                const icon = this.querySelector('.material-symbols-outlined');
                
                if (input.type === 'password') {
                    input.type = 'text';
                    icon.textContent = 'visibility_off';
                } else {
                    input.type = 'password';
                    icon.textContent = 'visibility';
                }
            });
        });

        // Simple role selection toggle
        const roles = document.querySelectorAll('section.flex.gap-xs button');
        roles.forEach(role => {
            role.addEventListener('click', () => {
                roles.forEach(r => {
                    r.classList.remove('bg-brand-orange', 'text-white');
                    r.classList.add('bg-white', 'border', 'border-brand-orange', 'text-brand-orange');
                });
                role.classList.remove('bg-white', 'border', 'border-brand-orange', 'text-brand-orange');
                role.classList.add('bg-brand-orange', 'text-white');
            });
        });
    </script>
